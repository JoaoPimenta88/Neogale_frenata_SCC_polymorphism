## POOL-SEQ ASSOCIATION MAPPING --

##1. CREATE INPUT FILES

##1.1. MERGE BAMS PER PHENOTYPE/SUBSPECIES WITH SAMTOOLS:
## merge of bams from individuals w/ same phenotype into one

samtools merge -@ 5 -b Brown_samples_east_frenata.txt Brown_samples_east_frenata.bam
samtools merge -@ 5 -b White_samples_east_frenata.txt White_samples_east_frenata.bam
samtools merge -@ 5 -b Brown_samples_west_frenata.txt Brown_samples_west_frenata.bam
samtools merge -@ 5 -b White_samples_west_frenata.txt White_samples_west_frenata.bam

#1.2. CREATE MPILEUP WITH SAMTOOLS FOR EACH TRANSITION REGION: 
	samtools mpileup -B -q 20 -Q 20 --positions ../DNA_damage/frenata_final_east_vison_min5-sites.damfilt.txt --reference ../references/vison/GCF_020171115.1_ASM_NN_V1_genomic.fna ../0_merged/Brown_samples_east_frenata.bam ../0_merged/White_samples_east_frenata.bam > frenata_east_final_min5.vison.damfilt.mpileup
	samtools mpileup -B -q 20 -Q 20 --positions ../DNA_damage/frenata_final_west_vison_min5-sites.damfilt.txt --reference ../references/vison/GCF_020171115.1_ASM_NN_V1_genomic.fna ../0_merged/Brown_samples_west_frenata.bam ../0_merged/White_samples_west_frenata.bam > frenata_west_final_min5.vison.damfilt.mpileup

#1.3. CREATE SYNC WITH POPOOLATION2:
##Create popoolation2 input file .sync from samtools mpileup.

 	java -ea -Xmx7g -jar ../bin/popoolation2_1201/mpileup2sync.jar --input frenata_east_final_min5.vison.damfilt.mpileup --output frenata_east_final_min5.vison.damfilt.sync --fastq-type sanger --min-qual 20 --threads 5
 	java -ea -Xmx7g -jar ../bin/popoolation2_1201/mpileup2sync.jar --input frenata_west_final_min5.vison.damfilt.mpileup --output frenata_west_final_min5.vison.damfilt.sync --fastq-type sanger --min-qual 20 --threads 5

#1.4. FIX MPILEUP TO INDEL FILTERING:

sed 's/\t\t/\t!\t!/g' frenata_east_final_min5.vison.damfilt.mpileup > frenata_east_final_min5.vison.damfilt.fixed.mpileup
sed 's/\t\t/\t!\t!/g' frenata_west_final_min5.vison.damfilt.mpileup > frenata_west_final_min5.vison.damfilt.fixed.mpileup

#1.5. INDENTIFY INDELS WITH POPOOLATION2:
#Indentifies indels and a window of 5 bp around it that will be masked in the following step.

perl ../bin/popoolation2_1201/indel_filtering/identify-indel-regions.pl --input frenata_east_final_min5.vison.damfilt.fixed.mpileup --output frenata_east_final_min5.vison.damfilt.fixed.indels.gtf --indel-window 5
perl ../bin/popoolation2_1201/indel_filtering/identify-indel-regions.pl --input frenata_west_final_min5.vison.damfilt.fixed.mpileup --output frenata_west_final_min5.vison.damfilt.fixed.indels.gtf --indel-window 5

#1.6. FILTER INDELS IN SYNC WITH POPOOLATION2:
##Masks indels identified in the previous step.

perl ../bin/popoolation2_1201/indel_filtering/filter-sync-by-gtf.pl --gtf frenata_east_final_min5.vison.damfilt.fixed.indels.gtf --input frenata_east_final_min5.vison.damfilt.sync --output frenata_east_final_min5.vison.damfilt.filtered.sync
perl ../bin/popoolation2_1201/indel_filtering/filter-sync-by-gtf.pl --gtf frenata_west_final_min5.vison.damfilt.fixed.indels.gtf --input frenata_west_final_min5.vison.damfilt.sync --output frenata_west_final_min5.vison.damfilt.filtered.sync


#2. FST & FISHER EXACT TEST COMPUTATION

#2.1. FST IN SLIDING WINDOW:

perl ../bin/popoolation2_1201/fst-sliding.pl --input frenata_east_final_min5.vison.damfilt.filtered.sync --output frenata_east_final_min5.damfilt.vison.filtered.50k25kcov5_mincount2.fst --karlsson-fst --min-covered-fraction 0.25 --min-count 2 --min-coverage 5 --max-coverage 33,33 --pool-size 22:22 --window-size 50000 --step-size 25000

#2.4. FISHER EXACT TEST PER WINDOW:

perl ../bin/popoolation2_1201/fisher-test.pl --input frenata_east_final_min5.vison.damfilt.filtered.sync --output frenata_east_final_min5.damfilt.vison.filtered.20k20kcov5_mincount2.fet --min-count 2 --min-coverage 5 --max-coverage 33  --window-size 20000 --step-size 20000 --window-summary-method median
