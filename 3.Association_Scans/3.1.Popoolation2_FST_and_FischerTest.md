## POOL-SEQ ASSOCIATION MAPPING --

##1. CREATE INPUT FILES

##1.1. MERGE BAMS PER PHENOTYPE/SUBSPECIES WITH SAMTOOLS:
## merge of bams from individuals w/ same phenotype into one

samtools merge -@ 5 -b Brown_samples_east_frenata.txt Brown_east-assembly50k_finaldataset.bam
samtools merge -@ 5 -b White_samples_east_frenata.txt White_east-assembly50k_finaldataset.bam
samtools merge -@ 5 -b Brown_samples_west_frenata.txt Brown_west-assembly50k_finaldataset.bam
samtools merge -@ 5 -b White_samples_west_frenata.txt White_west-assembly50k_finaldataset.bam

#1.2. CREATE MPILEUP WITH SAMTOOLS FOR EACH TRANSITION REGION: 
samtools mpileup -B -q 20 -Q 20 --positions /data/evochange/jpimenta/frenata/low_coverage/Analysis/DNA_damage/frenata_final_east_assembly50k_min5-sites.damfilt.txt --reference /data/evochange/jpimenta/References/nivalis/Mustela-nivalis_denovo-assembly-1700M-50k.fasta ../../0_merged/Brown_east-assembly50k_finaldataset.bam ../../0_merged/White_east-assembly50k_finaldataset.bam > frenata_east_final_min5.assembly.mpileup
samtools mpileup -B -q 20 -Q 20 --positions /data/evochange/jpimenta/frenata/low_coverage/Analysis/DNA_damage/frenata_final_west_assembly50k_min5-sites.damfilt.txt --reference /data/evochange/jpimenta/References/nivalis/Mustela-nivalis_denovo-assembly-1700M-50k.fasta ../../0_merged/Brown_west-assembly50k_finaldataset.bam ../../0_merged/White_west-assembly50k_finaldataset.bam > frenata_west_final_min5.assembly.mpileup

#1.3. CREATE SYNC WITH POPOOLATION2:
##Create popoolation2 input file .sync from samtools mpileup.

java -ea -Xmx7g -jar /data/evochange/bin/popoolation2_1201/mpileup2sync.jar --input frenata_east_final_min5.assembly.mpileup --output frenata_east_final_min5.damfilt.assembly.sync --fastq-type sanger --min-qual 20 --threads 5
java -ea -Xmx7g -jar /data/evochange/bin/popoolation2_1201/mpileup2sync.jar --input frenata_west_final_min5.assembly.mpileup --output frenata_west_final_min5.damfilt.assembly.sync --fastq-type sanger --min-qual 20 --threads 5

#1.4. FIX MPILEUP TO INDEL FILTERING:

sed 's/\t\t/\t!\t!/g' frenata_east_final_min5.assembly.mpileup > frenata_east_final_min5.assembly.fixed.mpileup
sed 's/\t\t/\t!\t!/g' frenata_west_final_min5.assembly.mpileup > frenata_west_final_min5.assembly.fixed.mpileup

#1.5. INDENTIFY INDELS WITH POPOOLATION2:
#Indentifies indels and a window of 5 bp around it that will be masked in the following step.

perl /data/evochange/bin/popoolation2_1201/indel_filtering/identify-indel-regions.pl --input frenata_east_final_min5.assembly.fixed.mpileup --output frenata_east_final_min5.damfilt.assembly.indels.gtf --indel-window 5
perl /data/evochange/bin/popoolation2_1201/indel_filtering/identify-indel-regions.pl --input frenata_west_final_min5.assembly.fixed.mpileup --output frenata_west_final_min5.damfilt.assembly.indels.gtf --indel-window 5

#1.6. FILTER INDELS IN SYNC WITH POPOOLATION2:
##Masks indels identified in the previous step.

perl /data/evochange/bin/popoolation2_1201/indel_filtering/filter-sync-by-gtf.pl --gtf frenata_east_final_min5.damfilt.assembly.indels.gtf --input frenata_east_final_min5.damfilt.assembly.sync --output frenata_east_final_min5.damfilt.assembly.filtered.sync
perl /data/evochange/bin/popoolation2_1201/indel_filtering/filter-sync-by-gtf.pl --gtf frenata_west_final_min5.damfilt.assembly.indels.gtf --input frenata_west_final_min5.damfilt.assembly.sync --output frenata_west_final_min5.damfilt.assembly.filtered.sync


#2. FST & FISHER EXACT TEST COMPUTATION

#This are only example commands: you can play w/ the filtering as much as you want. In an initial approach, I don't think is useful to use --min-covered-fraction flag.
#Check the software help for details on each flag
#The most important flags to play with are --window-size and --step-size. DOn't use --min-count lower than 3. Remember --min-coverage and --max-coverage are per population, not for the total dataset; try lower values at first, maybe don't even include --max-coverage at first. Usually I use --max-coverage 3 times the population genome-wide average. --pool-size is haploid number of chromosomes, so 2N, and order corresponds to the order of bam file input in mpileup. Same for --max-coverage (if different between pops)

#2.1. FST IN SLIDING WINDOW:

perl /data/evochange/bin/popoolation2_1201/fst-sliding.pl --input frenata_east_final_min5.damfilt.assembly.filtered.sync --output frenata_east_final_min5.damfilt.assembly.filtered.50k25kcov5_mincount2.fst --karlsson-fst --min-covered-fraction 0.25 --min-count 2 --min-coverage 5 --max-coverage 33,33 --pool-size 22:22 --window-size 50000 --step-size 25000

#2.4. FISHER EXACT TEST PER WINDOW:

perl /data/evochange/bin/popoolation2_1201/fisher-test.pl --input frenata_east_final_min5.damfilt.assembly.filtered.sync --output frenata_east_final_min5.damfilt.assembly.filtered.20k20kcov5_mincount2.fet --min-count 2 --min-coverage 5 --max-coverage 33  --window-size 20000 --step-size 20000 --window-summary-method median
