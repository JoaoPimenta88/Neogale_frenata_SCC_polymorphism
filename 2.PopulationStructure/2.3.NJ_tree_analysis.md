## NJ tree based on ngsDist genetic distances

## 1. Estimate genotype probabilities with ANGSD:

angsd -b samples_frenata.txt -ref GCF_020171115.1_ASM_NN_V1_genomic.fna -out gp_min11_maf0.020_maxcov132_maf0.020.damfilt -sites snps_frenata_min22_maxcov132_maf0.020.damfilt.20k.txt -rf chr_frenata_min22_maxcov132_maf0.020.damfilt.20k.txt -uniqueOnly 1 -remove_bads 1 -minQ 20 -minMapQ 20 -doCounts 1 -GL 1 -doMajorMinor 1 -doMaf 1 -doGeno 8 -doPost 1

## 2. Run ngsDist to estimate genetic distances
# with bootstrap:

ngsDist --geno gp_min11_maf0.020_maxcov132_maf0.020.damfilt.geno.gz --probs --labels pop_frenata_bam.txt --n_ind 44 --n_sites 123718 --n_boot_rep 100 --boot_block_size 25 --out gp_min11_maf0.020_maxcov132_maf0.020.damfilt.100boot.ngsdist &> log-gp_min11_maf0.020_maxcov132_maf0.020.damfilt-probs-100boot.txt

## 3. Run FastME to produce trees:
## 3.2. Genotype likelihoods:

#4. with bootstrap:

fastme -i gp_min11_maf0.020_maxcov132_maf0.020.damfilt.100boot.ngsdist -m BioNJ -D 101 -o gp_min11_maf0.020_maxcov132_maf0.020.damfilt.100boot.nwk

#5. RaXML

raxmlHPC-PTHREADS-SSE3 -f b -t gp_min11_maf0.020_maxcov132_maf0.020.damfilt.100boot.main.nwk -z gp_min11_maf0.020_maxcov132_maf0.020.damfilt.100boot.boots.nwk -m GTRCAT -n gp_min11_maf0.020_maxcov132_maf0.020.damfilt.final_boot
