## POOL-SEQ ASSOCIATION MAPPING

# 1. CREATE INPUT FILES

# 1.1. CREATE MPILEUP WITH SAMTOOLS: 

	samtools mpileup -B -q 20 -Q 20 --positions ../DNA_damage/frenata_final_east_vison_min5-sites.damfilt.txt --reference ../references/vison/GCF_020171115.1_ASM_NN_V1_genomic.fna ../0_merged/mc1r_brown_east.bam > /tmp/Brown_east_mc1r.mpileup
	samtools mpileup -B -q 20 -Q 20 --positions ../DNA_damage/frenata_final_east_vison_min5-sites.damfilt.txt --reference ../references/vison/GCF_020171115.1_ASM_NN_V1_genomic.fna ../0_merged/mc1r_white_east.bam > /tmp/White_east_mc1r.mpileup
	samtools mpileup -B -q 20 -Q 20 --positions ../DNA_damage/frenata_final_west_vison_min5-sites.damfilt.txt --reference ../references/vison/GCF_020171115.1_ASM_NN_V1_genomic.fna ../0_merged/mc1r_brown_west.bam > /tmp/Brown_west_mc1r.mpileup
	samtools mpileup -B -q 20 -Q 20 --positions ../DNA_damage/frenata_final_west_vison_min5-sites.damfilt.txt --reference ../references/vison/GCF_020171115.1_ASM_NN_V1_genomic.fna ../0_merged/mc1r_white_west.bam > /tmp/White_west_mc1r.mpileup

# 1.2. FIX MPILEUP TO INDEL FILTERING:

	sed 's/\t\t/\t!\t!/g' /tmp/Brown_east_mc1r.mpileup > /tmp/Brown_east_mc1r.fixed.mpileup
	sed 's/\t\t/\t!\t!/g' /tmp/White_east_mc1r.mpileup > /tmp/White_east_mc1r.fixed.mpileup
	sed 's/\t\t/\t!\t!/g' /tmp/Brown_west_mc1r.mpileup > /tmp/Brown_west_mc1r.fixed.mpileup
	sed 's/\t\t/\t!\t!/g' /tmp/White_west_mc1r.mpileup > /tmp/White_west_mc1r.fixed.mpileup

# 1.3. INDENTIFY INDELS:

	perl ../popoolation_1.2.3/basic-pipeline/identify-genomic-indel-regions.pl --input /tmp/Brown_east_mc1r.fixed.mpileup --output /tmp/Brown_east_mc1r.indels.gtf --indel-window 5
	perl ../popoolation_1.2.3/basic-pipeline/identify-genomic-indel-regions.pl --input /tmp/White_east_mc1r.fixed.mpileup --output /tmp/White_east_mc1r.indels.gtf --indel-window 5
	perl ../popoolation_1.2.3/basic-pipeline/identify-genomic-indel-regions.pl --input /tmp/Brown_west_mc1r.fixed.mpileup --output /tmp/Brown_west_mc1r.indels.gtf --indel-window 5
	perl ../popoolation_1.2.3/basic-pipeline/identify-genomic-indel-regions.pl --input /tmp/White_west_mc1r.fixed.mpileup --output /tmp/White_west_mc1r.indels.gtf --indel-window 5

# 1.4. FILTER INDELS:

	perl ../popoolation_1.2.3/basic-pipeline/filter-pileup-by-gtf.pl --input /tmp/Brown_east_mc1r.fixed.mpileup --gtf /tmp/Brown_east_mc1r.indels.gtf --output /tmp/Brown_east_mc1r.filtered.pileup
	perl ../popoolation_1.2.3/basic-pipeline/filter-pileup-by-gtf.pl --input /tmp/White_east_mc1r.fixed.mpileup --gtf /tmp/White_east_mc1r.indels.gtf --output /tmp/White_east_mc1r.filtered.pileup
	perl ../popoolation_1.2.3/basic-pipeline/filter-pileup-by-gtf.pl --input /tmp/Brown_west_mc1r.fixed.mpileup --gtf /tmp/Brown_west_mc1r.indels.gtf --output /tmp/Brown_west_mc1r.filtered.pileup
	perl ../popoolation_1.2.3/basic-pipeline/filter-pileup-by-gtf.pl --input /tmp/White_west_mc1r.fixed.mpileup --gtf /tmp/White_west_mc1r.indels.gtf --output /tmp/White_west_mc1r.filtered.pileup

# 2. Popoolation summary statistics estimation
# 2.1. Estimation of sumstats in windows

	perl ../popoolation_1.2.3/Variance-sliding.pl --measure D --input White_east_mc1r.filtered.pileup --output /tmp/White_east_mc1r.filtered.D_w50k_s25k --pool-size 22 --min-count 2 --min-coverage 5 --min-covered-fraction 0.25 --window-size 50000 --step-size 25000 --fastq-type sanger
	perl ../popoolation_1.2.3/Variance-sliding.pl --measure D --input Brown_east_mc1r.filtered.pileup --output /tmp/Brown_east_mc1r.filtered.D_w50k_s25k --pool-size 22 --min-count 2 --min-coverage 5 --min-covered-fraction 0.25 --window-size 50000 --step-size 25000 --fastq-type sanger
	perl ../popoolation_1.2.3/Variance-sliding.pl --measure D --input White_west_mc1r.filtered.pileup --output /tmp/White_west_mc1r.filtered.D_w50k_s25k --pool-size 22 --min-count 2 --min-coverage 5 --min-covered-fraction 0.25 --window-size 50000 --step-size 25000 --fastq-type sanger
	perl ../popoolation_1.2.3/Variance-sliding.pl --measure D --input Brown_west_mc1r.filtered.pileup --output /tmp/Brown_west_mc1r.filtered.D_w50k_s25k --pool-size 22 --min-count 2 --min-coverage 5 --min-covered-fraction 0.25 --window-size 50000 --step-size 25000 --fastq-type sanger
